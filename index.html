// Character set: space = transparent, @ = brightest
const CHARSETS = {
  standard: ' .,:;+*=#@',
  blocks:   ' ░▒▓█',
  minimal:  ' .:@'
};

// Map brightness [0–255] → single character
function mapBrightnessToChar(brightness, charset = 'standard') {
  const chars = CHARSETS[charset];
  const index = Math.floor((brightness / 255) * (chars.length - 1));
  return chars[Math.max(0, Math.min(chars.length - 1, index))];
}

// Plasma: four overlapping sine waves → [0, 1]
function computePlasma(col, row, t) {
  const v1 = Math.sin(col * 0.10 + t);
  const v2 = Math.sin(row * 0.10 + t * 0.70);
  const v3 = Math.sin((col + row) * 0.07 + t * 0.90);
  const v4 = Math.sin(Math.sqrt(col*col + row*row) * 0.08 - t);
  return (v1 + v2 + v3 + v4 + 4) / 8; // → [0, 1]
}
